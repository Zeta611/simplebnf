\tl_new:N \l__simplebnf_lhs_tl
\tl_new:N \l__simplebnf_matched_relation_tl
\tl_new:N \l__simplebnf_rhs_tl

\seq_new:N \l__simplebnf_alternative_seq
\bool_new:N \l__simplebnf_first_alternative_bool
